### Task 1: LAN mode "not found" fix

**Background:** `npm run dev:lan` starts Vite on `0.0.0.0`. The backend is on `localhost:8765`. When phone opens `http://192.168.x.x:5173`, Vue Router serves the app fine. But when `DZMM_FRONTEND_DIST` is set (packaged app), the backend serves the frontend at `http://192.168.x.x:8765`. On the phone, navigating to any route like `/#/sessions` works, but direct hash-less URLs fail.

The actual problem reported is "not found" — most likely the phone is hitting `http://ip:8765/` (no hash) and getting 404 from FastAPI because no route matches `/`. The `StaticFiles(html=True)` should catch this, but it only works if mounted AFTER all API routes — verify this is the case.

**Additionally:** There may be a CORS or Content-Type issue for non-Tauri browsers.

**Files:**
- Modify: `backend/src/dzmm/main.py`

- [ ] **Step 1: Diagnose by reading `main.py`**

Read `backend/src/dzmm/main.py`. Confirm the static file mount is the LAST thing added (`app.mount("/", ...)` happens after all `app.include_router(...)` calls).

Current mount code:
```python
dist = os.environ.get("DZMM_FRONTEND_DIST")
if dist and Path(dist).is_dir():
    app.mount("/", StaticFiles(directory=dist, html=True), name="frontend")
```

`html=True` means FastAPI's `StaticFiles` will serve `index.html` for 404s within the mount. This should handle SPA routing. If the phone still gets "not found", the issue is something else.

- [ ] **Step 2: Add explicit index.html route before the static mount**

Add an explicit GET `/` route that returns `index.html` directly, before the static mount:

```python
from fastapi.responses import FileResponse

if dist and Path(dist).is_dir():
    index_html = Path(dist) / "index.html"

    @app.get("/")
    async def serve_index():
        return FileResponse(str(index_html))

    app.mount("/", StaticFiles(directory=dist, html=True), name="frontend")
```

This ensures that `GET /` always returns `index.html` even if `StaticFiles` has a race with API routes.

### Task 2: NPC profile auto-enrichment (background sub-agent)

**Background:** When the GM emits `<npc_update name="张三">` with only a name and one-line description, the NPC has an empty `purpose` and `archetype`. The NPC detail dialog shows very little info. A small background LLM call can fill this in within seconds, without blocking the main turn.

**Files:**
- Create: `backend/src/dzmm/service/npc_enrichment.py`
- Modify: `backend/src/dzmm/service/state_apply/npc.py`
- Modify: `backend/src/dzmm/service/state_apply/_impl.py`

- [ ] **Step 1: Create `npc_enrichment.py`**

```python
# backend/src/dzmm/service/npc_enrichment.py
"""Background NPC profile enrichment.

After a new NPC is created with minimal info (name + brief description only),
fire a small LLM call to fill in purpose and archetype. Runs as a background
asyncio task — does not block the turn response.
"""
import asyncio
import logging

from sqlalchemy.ext.asyncio import AsyncSession, async_sessionmaker
from sqlalchemy import select

from dzmm.db.models import ModelConfig, NPC, Session as GameSession
from dzmm.models.client import GenerationParams
from dzmm.models.factory import build_client

log = logging.getLogger(__name__)


async def enrich_npc_stub(
    session_maker: async_sessionmaker[AsyncSession],
    session_id: int,
    npc_id: int,
) -> None:
    """Fill in purpose/archetype for a stub NPC. Called as a background task."""
    async with session_maker() as s:
        npc = await s.get(NPC, npc_id)
        if npc is None:
            return
        # Only enrich stubs (no purpose AND no archetype)
        if npc.purpose or npc.archetype:
            return

        sess = await s.get(GameSession, session_id)
        if sess is None:
            return
        cfg = await s.get(ModelConfig, sess.gm_model_config_id)
        if cfg is None:
            return

        from dzmm.db.models import World
        world = await s.get(World, sess.world_id)
        world_snippet = (world.content_md or "")[:300] if world else ""

        client = build_client(cfg)
        messages = [
            {
                "role": "system",
                "content": (
                    f"世界观概要：{world_snippet}\n\n"
                    "你是世界观设计师。为以下 NPC 补充 purpose 和 archetype。\n"
                    "只输出 JSON：{\"purpose\":\"...\",\"archetype\":\"...\"}\n"
                    "purpose：这个 NPC 在故事中的作用（≤30字）\n"
                    "archetype：角色原型，如 导师/反派/盟友/商人/信使（≤10字）"
                ),
            },
            {
                "role": "user",
                "content": f"NPC：{npc.name}\n描述：{npc.description or '（无）'}",
            },
        ]

        chunks: list[str] = []
        try:
            async for ch in client.stream(
                messages, GenerationParams(max_tokens=100, temperature=0.7)
            ):
                if ch.delta:
                    chunks.append(ch.delta)
        except Exception as e:
            log.debug("NPC enrichment LLM error for %s: %s", npc.name, e)
            return

        raw = "".join(chunks).strip()
        try:
            import json, re
            cleaned = re.sub(r"^```(?:json)?\s*|\s*```$", "", raw, flags=re.DOTALL).strip()
            data = json.loads(cleaned)
            if data.get("purpose"):
                npc.purpose = str(data["purpose"])[:200]
            if data.get("archetype"):
                npc.archetype = str(data["archetype"])[:60]
            await s.commit()
            log.debug("Enriched NPC %s: purpose=%s archetype=%s", npc.name, npc.purpose, npc.archetype)
        except Exception as e:
            log.debug("NPC enrichment parse error for %s: %s raw=%r", npc.name, e, raw[:100])
```

```python
# Add to backend/tests/test_state_apply.py
@pytest.mark.asyncio
async def test_npc_update_triggers_enrichment_task(engine):
    """Creating a stub NPC schedules a background task (we verify the task is created,
    not its result — enrichment is fire-and-forget)."""
    import asyncio
    from unittest.mock import patch, AsyncMock
    from dzmm.service.state_apply._impl import apply_tags
    from dzmm.parsing.events import TagComplete

    tasks_created = []
    original_create_task = asyncio.create_task

    def mock_create_task(coro, *args, **kwargs):
        tasks_created.append(coro.__name__ if hasattr(coro, '__name__') else str(type(coro)))
        # Cancel immediately to avoid side effects
        t = original_create_task(asyncio.sleep(0))
        coro.close()
        return t

    async with AsyncSession(engine) as s:
        sess = GameSession(name="t", world_id=1, character_id=1,
                           gm_model_config_id=1, summarizer_model_config_id=1)
        s.add(sess)
        await s.flush()

        tag = TagComplete(name="npc_update", attrs={"name": "王五"}, content="")
        with patch("asyncio.create_task", side_effect=mock_create_task):
            fake_maker = AsyncMock()
            await apply_tags(s, sess.id, 1, [tag], session_maker=fake_maker)
        await s.commit()

    # A background task should have been scheduled
    assert len(tasks_created) > 0
```
